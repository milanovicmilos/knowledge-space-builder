pythonimport torch
import torch.nn as nn

class SparseKnowledgeVAE(nn.Module):
    def __init__(self, n_questions=120, latent_dim=10):
        super().__init__()
        
        # Encoder ignoriše NA vrednosti
        self.encoder = nn.Sequential(
            nn.Linear(n_questions, 64),
            nn.ReLU(),
            nn.Dropout(0.3)
        )
        self.mu = nn.Linear(64, latent_dim)
        self.logvar = nn.Linear(64, latent_dim)
        
        # Decoder rekonstruiše prostor znanja
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, 64),
            nn.ReLU(),
            nn.Linear(64, n_questions),
            nn.Sigmoid()
        )
    
    def forward(self, x, mask):
        # mask = 1 gde postoje vrednosti, 0 za NA
        x_masked = x * mask
        
        # Encoding
        h = self.encoder(x_masked)
        mu, logvar = self.mu(h), self.logvar(h)
        
        # Reparametrization
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        z = mu + eps * std
        
        # Decoding
        reconstruction = self.decoder(z)
        
        return reconstruction, mu, logvar, z

# Loss funkcija koja ignoriše NA
def masked_loss(recon, target, mask):
    # Računaj loss samo za poznate vrednosti
    bce = nn.BCELoss(reduction='none')
    loss = bce(recon, target) * mask
    return loss.sum() / mask.sum()
3. Graph Neural Networks (GNN) - CUTTING EDGE
Modeluj studente i pitanja kao graf:
pythonimport torch_geometric as pyg
from torch_geometric.nn import GCNConv

class BipartiteKnowledgeGNN(nn.Module):
    def __init__(self, n_students, n_questions, hidden_dim=64):
        super().__init__()
        
        # Student i Question embeddings
        self.student_embed = nn.Embedding(n_students, hidden_dim)
        self.question_embed = nn.Embedding(n_questions, hidden_dim)
        
        # Graph convolutions
        self.conv1 = pyg.nn.SAGEConv(hidden_dim, hidden_dim)
        self.conv2 = pyg.nn.SAGEConv(hidden_dim, hidden_dim)
        
        # Prediction
        self.predict = nn.Linear(hidden_dim * 2, 1)
    
    def forward(self, edge_index, student_ids, question_ids):
        # Create bipartite graph
        x_students = self.student_embed(student_ids)
        x_questions = self.question_embed(question_ids)
        
        # Apply GNN (rade samo sa postojećim vezama)
        x = torch.cat([x_students, x_questions], dim=0)
        x = self.conv1(x, edge_index).relu()
        x = self.conv2(x, edge_index).relu()
        
        # Predict student-question interaction
        student_repr = x[student_ids]
        question_repr = x[question_ids]
        pred = self.predict(torch.cat([student_repr, question_repr], dim=1))
        
        return pred.sigmoid()

import torch
from torch import nn

class DKTModel(nn.Module):
    def __init__(self, n_questions, hidden_size=200):
        super().__init__()
        self.lstm = nn.LSTM(n_questions * 2, hidden_size, batch_first=True)
        self.fc = nn.Linear(hidden_size, n_questions)
    
    def forward(self, x):
        # x: (batch, seq_len, n_questions * 2)  # question + response
        out, _ = self.lstm(x)
        pred = self.fc(out)
        return torch.sigmoid(pred)

# 3. Transformer za long-range dependencies
class KnowledgeTransformer(nn.Module):
    def __init__(self, n_questions, d_model=128, nhead=8):
        super().__init__()
        self.embedding = nn.Linear(n_questions * 2, d_model)
        self.transformer = nn.TransformerEncoder(
            nn.TransformerEncoderLayer(d_model, nhead),
            num_layers=6
        )
        self.output = nn.Linear(d_model, n_questions)
    
    def forward(self, x, mask):
        x = self.embedding(x)
        x = self.transformer(x, src_key_padding_mask=~mask)
        return torch.sigmoid(self.output(x))

def build_poset_from_sets(sets):
    # sets: list of frozenset of items
    unique = sorted(list(set(sets)), key=lambda s: (len(s), tuple(sorted(s))))
    # map to string keys for JSON
    key_for = {s: '{' + ', '.join(sorted(s)) + '}' for s in unique}

    # Compute cover relation (immediate supersets)
    children = {key_for[s]: [] for s in unique}
    n = len(unique)
    for i, a in enumerate(unique):
        for b in unique[i+1:]:
            if a.issubset(b):
                # check if there's an intermediate c with a ⊂ c ⊂ b
                has_mid = False
                for c in unique[i+1:]:
                    if a.issubset(c) and c.issubset(b) and c != b:
                        if a != c:
                            has_mid = True
                            break
                if not has_mid:
                    children[key_for[a]].append(key_for[b])
    return children

# Graf ima 120 čvorova (pitanja), ne 127k stanja

def build_prerequisite_graph(pred_probs, item_cols, threshold=0.85):
    """
    Za svaki par pitanja (A, B), proveri:
    Ako student zna B, koliko često zna i A?
    Ako > threshold, onda A je prerequisite za B
    """
    n_items = len(item_cols)
    binary = pred_probs >= 0.6
    
    prereqs = {}  # prereqs[B] = lista A gde je A prerequisite za B
    
    for b in range(n_items):
        prereqs[item_cols[b]] = []
        knows_b = binary[:, b]  # studenti koji znaju B
        
        if knows_b.sum() < 10:  # skip ako premalo podataka
            continue
            
        for a in range(n_items):
            if a == b:
                continue
            
            knows_a = binary[:, a]
            # Od onih koji znaju B, koliko % zna i A?
            implication_rate = (knows_b & knows_a).sum() / knows_b.sum()
            
            if implication_rate > threshold:
                prereqs[item_cols[b]].append(item_cols[a])
    
    # Transitive reduction - ukloni redundantne veze
    G = nx.DiGraph()
    for b, prereq_list in prereqs.items():
        for a in prereq_list:
            G.add_edge(a, b)
    
    # Ukloni tranzitivne veze (ako A->C i C->B, ukloni A->B)
    G_reduced = nx.transitive_reduction(G)
    
    return G_reduced

def cluster_knowledge_states(pred_probs, n_clusters=50):
    """
    Grupiši 220k studenata u ~50 reprezentativnih knowledge states
    """
    clustering = AgglomerativeClustering(n_clusters=n_clusters)
    labels = clustering.fit_predict(pred_probs)
    
    # Za svaki klaster, uzmi prosečan response pattern
    cluster_centers = []
    for i in range(n_clusters):
        mask = labels == i
        center = pred_probs[mask].mean(axis=0)
        cluster_centers.append(center >= 0.6)  # threshold
    
    # Sada imas samo 50 stanja umesto 127k
    return cluster_centers

    """Generiši SVE validne knowledge states koji poštuju prerequisite-e"""
    from itertools import combinations
    
    def is_valid_state(items_set):
        """State je validan ako sadrži sve prerequisite-e"""
        for item in items_set:
            prereqs = set(prereq_graph.get(item, []))
            if not prereqs.issubset(items_set):
                return False
        return True
    
    valid_states = [frozenset()]  # Prazan skup
    
    # Generiši sve kombinacije do max_size
    for size in range(1, min(len(item_cols), max_size) + 1):

            state = frozenset(combo)
            if is_valid_state(state):
                valid_states.append(state)
    
    # Dodaj kompletan skup ako je validan
    full_set = frozenset(item_cols)
    if is_valid_state(full_set):
        valid_states.append(full_set)
    
    return valid_states

pythondef build_poset_from_sets(sets):
    """Build cover relations for lattice"""
    unique_list = list(dict.fromkeys(sets))
    key_for = {s: '{' + ', '.join(sorted(s)) + '}' for s in unique_list}

    element_index = {}
    for idx, s in enumerate(unique_list):
        for e in s:
            element_index.setdefault(e, set()).add(idx)

    all_indices = set(range(len(unique_list)))
    children = {key_for[s]: [] for s in unique_list}

    for i, A in enumerate(unique_list):
        if len(A) == 0:
            candidates = all_indices - {i}
        else:
            lists = [element_index.get(e, set()) for e in A]
            if not lists:
                candidates = set()
            else:
                candidates = set.intersection(*lists)
            candidates.discard(i)

        if not candidates:
            continue

        min_size = min(len(unique_list[j]) for j in candidates)
        minimal_supers = [j for j in candidates if len(unique_list[j]) == min_size]

        for j in minimal_supers:
            if A.issubset(unique_list[j]):
                children[key_for[A]].append(key_for[unique_list[j]])

    return children

pythondef closure_under_union_and_intersection(states, max_iterations=5, max_states=2000):
    """
    Compute closure with safety limits
    """
    changed = True
    states = set(states)
    iteration = 0
    
    while changed and iteration < max_iterations:
        if len(states) > max_states:
            logger.warning('Reached max_states limit (%d), stopping closure early', max_states)
            break
            
        changed = False
        new_sets = set()
        lst = list(states)
        L = len(lst)
        
        for i in range(L):
            for j in range(i, L):
                a = lst[i]
                b = lst[j]
                u = a.union(b)
                iv = a.intersection(b)
                
                if u not in states:
                    new_sets.add(u)
                if iv not in states:
                    new_sets.add(iv)
                    
                # Early stop if too many new sets
                if len(new_sets) > 500:
                    break
            if len(new_sets) > 500:
                break
                
        if new_sets:
            states |= new_sets
            changed = True
            iteration += 1
            logger.info('Closure iteration %d: added %d states (total: %d)', 
                       iteration, len(new_sets), len(states))
    
    logger.info('States after closure: %d (iterations: %d)', len(states), iteration)
    return states

pythondef build_empirical_knowledge_space(pred_probs, item_cols, threshold=0.6, min_support=5):
    """
    Build knowledge space from actual student response patterns
    """
    binary = pred_probs >= threshold
    
    # Extract unique knowledge states from students
    states = []
    for row in binary:
        state = frozenset([item_cols[i] for i, mastered in enumerate(row) if mastered])
        states.append(state)
    
    # Count frequency
    from collections import Counter
    state_counts = Counter(states)
    
    # Filter by min_support
    frequent_states = [s for s, count in state_counts.items() if count >= min_support]
    
    # Add empty set if missing
    if frozenset() not in frequent_states:
        frequent_states.insert(0, frozenset())
    
    logger.info(f'Empirical knowledge space: {len(frequent_states)} states from student data')
    
    return frequent_states

pythondef generate_valid_states_from_prereq(prereq_map, items, max_size=None, progress_log=True):
    """
    Generate all states over `items` that satisfy prerequisite constraints.
    With progress logging for large k.
    """
    from itertools import combinations
    import time

    item_set = list(items)
    n = len(item_set)
    if max_size is None:
        max_size = n

    def is_valid_state(state):
        for it in state:
            prereqs = set(prereq_map.get(it, []))
            if not prereqs.issubset(state):
                return False
        return True

    states = set()
    states.add(frozenset())
    
    start_time = time.time()
    total_checked = 0
    
    for size in range(1, min(max_size, n) + 1):
        size_start = time.time()
        checked_this_size = 0
        valid_this_size = 0
        
        for combo in combinations(item_set, size):
            s = frozenset(combo)
            checked_this_size += 1
            total_checked += 1
            
            if is_valid_state(s):
                states.add(s)
                valid_this_size += 1
        
        if progress_log:
            elapsed = time.time() - size_start
            logger.info(f'Size {size}: checked {checked_this_size:,} combos, '
                       f'found {valid_this_size} valid states in {elapsed:.2f}s '
                       f'(total states so far: {len(states)})')
    
    total_time = time.time() - start_time
    logger.info(f'Generated {len(states)} prerequisite-valid states '
               f'after checking {total_checked:,} combinations in {total_time:.2f}s')
    return states
